cmake_minimum_required(VERSION 3.16)
project(cubeworld_omega_rules LANGUAGES CXX)
add_executable(test_omega_rules tests/test_omega_rules.cpp)
target_compile_features(test_omega_rules PRIVATE cxx_std_17)
target_include_directories(test_omega_rules PRIVATE src)
if(MSVC)
  target_compile_options(test_omega_rules PRIVATE /W4)
else()
  target_compile_options(test_omega_rules PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME omega_rules COMMAND test_omega_rules)
